# Repository Implementation Example

This example shows how to implement a repository contract across multiple languages, following the same specification.

## Specification

We need a `UserRepository` that can:

- `findById(id): User | null`
- `save(user): User`
- `findByEmail(email): User | null`
- `delete(id): void`

Constraints:

- Email must be unique.
- IDs are generated by the database.
- Returned users are never mutated by the repository.

## Kotlin

```kotlin
package com.example.user.adapter

import com.example.user.domain.User
import com.example.user.domain.UserId
import com.example.user.domain.Email
import com.example.user.port.UserRepository

class SqlUserRepository(
    private val database: Database
) : UserRepository {

    override fun findById(id: UserId): User? =
        database.query(
            "SELECT id, email, name FROM users WHERE id = ?",
            id.value
        ).firstOrNull()?.toUser()

    override fun findByEmail(email: Email): User? =
        database.query(
            "SELECT id, email, name FROM users WHERE email = ?",
            email.value
        ).firstOrNull()?.toUser()

    override fun save(user: User): User {
        require(user.id == null) { "Cannot save a user that already has an ID" }

        val generatedId = database.insert(
            "INSERT INTO users (email, name) VALUES (?, ?)",
            user.email.value,
            user.name
        )

        return user.copy(id = UserId(generatedId))
    }

    override fun delete(id: UserId) {
        database.execute("DELETE FROM users WHERE id = ?", id.value)
    }

    private fun Row.toUser(): User = User(
        id = UserId(getLong("id")),
        email = Email(getString("email")),
        name = getString("name")
    )
}
```

## TypeScript

```typescript
import { User, UserId, Email } from '../domain'
import { UserRepository } from '../port'

export class SqlUserRepository implements UserRepository {
  constructor(private readonly database: Database) {}

  async findById(id: UserId): Promise<User | null> {
    const row = await this.database.query(
      'SELECT id, email, name FROM users WHERE id = ?',
      id.value
    ).first()
    return row ? this.toUser(row) : null
  }

  async findByEmail(email: Email): Promise<User | null> {
    const row = await this.database.query(
      'SELECT id, email, name FROM users WHERE email = ?',
      email.value
    ).first()
    return row ? this.toUser(row) : null
  }

  async save(user: User): Promise<User> {
    if (user.id !== null) {
      throw new Error('Cannot save a user that already has an ID')
    }

    const generatedId = await this.database.insert(
      'INSERT INTO users (email, name) VALUES (?, ?)',
      user.email.value,
      user.name
    )

    return user.withId(new UserId(generatedId))
  }

  async delete(id: UserId): Promise<void> {
    await this.database.execute(
      'DELETE FROM users WHERE id = ?',
      id.value
    )
  }

  private toUser(row: Row): User {
    return new User(
      new UserId(row.id),
      new Email(row.email),
      row.name
    )
  }
}
```

## Python

```python
from dataclasses import replace
from typing import Optional
from example.user.domain import User, UserId, Email
from example.user.port import UserRepository


class SqlUserRepository(UserRepository):
    def __init__(self, database: Database) -> None:
        self._database = database

    def find_by_id(self, id: UserId) -> Optional[User]:
        row = self._database.query(
            "SELECT id, email, name FROM users WHERE id = ?",
            id.value
        ).first()
        return self._to_user(row) if row else None

    def find_by_email(self, email: Email) -> Optional[User]:
        row = self._database.query(
            "SELECT id, email, name FROM users WHERE email = ?",
            email.value
        ).first()
        return self._to_user(row) if row else None

    def save(self, user: User) -> User:
        if user.id is not None:
            raise ValueError("Cannot save a user that already has an ID")

        generated_id = self._database.insert(
            "INSERT INTO users (email, name) VALUES (?, ?)",
            user.email.value,
            user.name
        )

        return replace(user, id=UserId(generated_id))

    def delete(self, id: UserId) -> None:
        self._database.execute(
            "DELETE FROM users WHERE id = ?",
            id.value
        )

    def _to_user(self, row: Row) -> User:
        return User(
            id=UserId(row["id"]),
            email=Email(row["email"]),
            name=row["name"]
        )
```

## Key Takeaways

1. **Port/Adapter pattern**: The repository interface lives in the domain/application layer; the SQL implementation is in the infrastructure layer.
2. **Immutability**: Returned domain objects are not mutated by the repository; a new instance is returned after `save`.
3. **Error handling**: Expected errors (duplicate email) are translated into domain errors; accidental errors raise exceptions.
4. **Language idioms**: Each implementation follows its language's conventions for naming, types, and mutation.
5. **Testability**: The repository depends on an abstract `Database` so tests can use an in-memory fake.
